You get a list cards of exactly 4 whole numbers, each between 1 and 9. Use every card exactly once, joined by the operators +, -, * and / in any order and with any brackets, to see whether the result can be exactly 24.

Division is true division, so 8 / 3 is not 2. A minus sign only works between two values; a card cannot be negated on its own, and two cards cannot be glued into one number such as 12. Return True if some arrangement gives exactly 24, and False if none does.

solution: |
  from fractions import Fraction

  def judge_point_24(cards):
      def can(nums):
          if len(nums) == 1:
              return nums[0] == 24
          for i in range(len(nums)):
              for j in range(i):
                  rest = [nums[k] for k in range(len(nums)) if k != i and k != j]
                  a, b = nums[i], nums[j]
                  options = [a + b, a - b, b - a, a * b]
                  if b != 0:
                      options.append(a / b)
                  if a != 0:
                      options.append(b / a)
                  for v in options:
                      if can(rest + [v]):
                          return True
          return False

      return can([Fraction(c) for c in cards])

  def judge_point_24(cards):
      from itertools import permutations, product
      from fractions import Fraction

      def f(op, x, y):
          if x is None or y is None:
              return None
          if op == '+':
              return x + y
          if op == '-':
              return x - y
          if op == '*':
              return x * y
          return None if y == 0 else x / y

      for p in permutations(cards):
          a, b, c, d = map(Fraction, p)
          for o1, o2, o3 in product('+-*/', repeat=3):
              shapes = [
                  f(o3, f(o2, f(o1, a, b), c), d),
                  f(o3, f(o1, a, f(o2, b, c)), d),
                  f(o2, f(o1, a, b), f(o3, c, d)),
                  f(o1, a, f(o3, f(o2, b, c), d)),
                  f(o1, a, f(o2, b, f(o3, c, d))),
              ]
              if any(v == 24 for v in shapes):
                  return True
      return False
